# Transformers 4.50+ hard-requires torch >= 2.5 and silently disables PyTorch
# when it finds 2.3.1, which breaks the WhisperX wav2vec2 aligner. 4.48.3 is the
# last line that fully supports torch 2.3.x, and it keeps Demucs 4.0.1 on the
# torchaudio 2.3.1 API it was released against.
transformers==4.48.3

# PyAV: av 11.x has no cp311 manylinux wheel and compiles against FFmpeg 6.0
# headers (AV_OPT_TYPE_CHLAYOUT failure). av 12.3.0 ships a cp311 wheel with
# FFmpeg bundled — no compilation at all.
av==12.3.0

# CTranslate2 4.5+ wheels are marked as requiring an executable stack, which
# glibc >= 2.41 (Debian trixie) refuses to load:
#   "cannot enable executable stack as shared object requires: Invalid argument"
# We pin 4.4.0 (also the ceiling whisperx 3.3.1 declares) AND pin the base image
# to bookworm/glibc 2.36 so the wheel loads.
ctranslate2==4.4.0
faster-whisper==1.1.0
whisperx==3.3.1

# WhisperX / pyannote.audio import paths pull in matplotlib at build-time smoke
